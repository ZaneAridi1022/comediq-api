venues:
    id             SERIAL
    name           TEXT
    address        TEXT
    city           TEXT
    neighbourhood  TEXT
    borough        TEXT
    type           TEXT
    contact        TEXT

    PRIMARY KEY: id
    UNIQUE: name, address, city, neighbourhood, borough, type, contact

venue_rooms:
    id         SERIAL
    venue_id   SERIAL
    name       TEXT

    PRIMARY KEY: id
    FOREIGN KEY: venue_id REFERENCES venues.id
    UNIQUE: venue_id, name

upcoming_show_instances:
    id                   SERIAL
    venue_room_id        SERIAL
    name                 STRING
    signup_instructions  STRING
    notes                STRING | NULL
    audience_cost        STRING | NULL
    comedian_cost        STRING | NULL
    stage_time           STRING | NULL
    host                 STRING | NULL
    sms                  STRING | NULL
    verified             STRING
    comedian_lineup      STRING[]
    start_time           DTZ
    end_time             DTZ
    active               BOOLEAN

    PRIMARY KEY: id
    FOREIGN KEY: venue_room_id REFERENCES venue_rooms.id

show_definitions:
    id                   SERIAL
    venue_room_id        SERIAL
    name                 STRING
    signup_instructions  STRING
    notes                STRING | NULL
    audience_cost        STRING | NULL
    comedian_cost        STRING | NULL
    stage_time           STRING | NULL
    host                 STRING | NULL
    sms                  STRING | NULL
    verified             STRING
    occurrence_rules     schedule_rule_ids: <SERIAL | NULL>[7]

    PRIMARY KEY: id
    FOREIGN KEY: venue_room_id REFERENCES venue_rooms.id

historical_show_instances:
    id              SERIAL
    venue_room_id   SERIAL
    name            STRING 
    host            STRING | NULL
    audience_cost   STRING | NULL
    comedian_cost   STRING | NULL
    stage_time      STRING | NULL
    comedian_lineup STRING[]
    start_time      DTZ
    end_time        DTZ

    PRIMARY KEY: id
    FOREIGN KEY: venue_room_id REFERENCES venue_rooms.id

performance_notes:
    id                  SERIAL
	comedian_id         UUID
	historical_show_id  SERIAL
	comedian_notes      STRING
	video_url           STRING

	PRIMARY KEY: id
	FOREIGN KEY: comedian_id REFERENCES (comedians table).id
	FOREIGN KEY: historical_show_id REFERENCES historical_show_instances.id

one_month_day_rule:
    id: SERIAL
    monthsPreparedInAdvance: int (default 1)
    onlyLastDayOfMonth: bool
    weeklyPolicy: [bool, bool, bool, bool, bool]
    StartTime: string
    EndTime: string

    PRIMARY KEY: id
